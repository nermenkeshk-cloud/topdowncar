// helper.js const keys = { up: 'KeyW', left: 'KeyA', right: 'KeyD', down: 'KeyS', boost: 'Space' }; function pressKey(code, duration = 80) { const eventDown = new KeyboardEvent('keydown', { code, bubbles: true }); const eventUp = new KeyboardEvent('keyup', { code, bubbles: true }); document.dispatchEvent(eventDown); setTimeout(() => document.dispatchEvent(eventUp), duration); } setInterval(() => { pressKey(keys.up, 100); pressKey(keys.right, 70); }, 500);
